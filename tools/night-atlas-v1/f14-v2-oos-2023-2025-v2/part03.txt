def preflight(manifest: Path, standards: Path, campaign_dir: Path, dependency_hashes: dict[str, str]):
    if manifest.resolve().parent != campaign_dir.resolve():
        raise AdmissionError("manifest must be the canonical manifest in the script campaign directory")
    rows = list(admitted_rows(manifest))
    for p in (
        standards / "GUARDIAN_RESEARCH_PROTOCOL_V1.md",
        standards / "TRADE_OBSERVATION_SCHEMA_V1.json",
        standards / "guardian_observation_v1.py",
    ):
        if not p.is_file():
            raise AdmissionError(f"missing research standard: {p}")
    oos_days = sum(
        OOS_START <= datetime.fromisoformat(r["date"]).replace(tzinfo=UTC) < END_EXCLUSIVE
        for r in rows
    )
    return {
        "status": "PREFLIGHT_ONLY",
        "id": ID,
        "dependency_sha256": dependency_hashes,
        "reference_engine_sha256": REFERENCE_ENGINE_SHA256,
        "index_rows_with_warmup": len(rows),
        "oos_index_rows": oos_days,
        "oos_start": OOS_START.isoformat(),
        "oos_end_exclusive": END_EXCLUSIVE.isoformat(),
        "oos_price_bars_decoded": False,
        "protected_2026_opened": False,
    }


def run(manifest: Path, standards: Path, campaign_dir: Path, dependency_hashes: dict[str, str]):
    output_dir = CANONICAL_OUTPUT_DIR
    lock_path = campaign_dir / CANONICAL_LOCK_NAME

    if output_dir.exists():
        raise AdmissionError(f"canonical output already exists: {output_dir}")

    # Metadata/index-only preflight. No market payload is decoded here.
    preflight(manifest, standards, campaign_dir, dependency_hashes)

    code_hash = sha256_file(Path(__file__).resolve())
    manifest_hash = canonical_manifest_sha256(manifest)
    spec = freeze_spec(code_hash, manifest_hash, dependency_hashes)
    spec_bytes = canonical_json_bytes(spec)
    spec_hash = hashlib.sha256(spec_bytes).hexdigest()

    lock = {
        "schema": 2,
        "id": ID,
        "status": "RESERVED",
        "phase": "RESERVED",
        "oos_start": OOS_START.isoformat(),
        "oos_end_exclusive": END_EXCLUSIVE.isoformat(),
        "canonical_output_dir": str(output_dir),
        "freeze_spec_sha256": spec_hash,
        "code_sha256": code_hash,
        "dependency_sha256": dependency_hashes,
        "reference_engine_sha256": REFERENCE_ENGINE_SHA256,
        "protected_2026_opened": False,
    }

    # Exclusive canonical reservation: a second process cannot claim the same OOS.
    try:
        with lock_path.open("x", encoding="utf-8", newline="\n") as f:
            f.write(canonical_json_bytes(lock).decode("utf-8"))
    except FileExistsError as exc:
        raise AdmissionError(f"OOS already reserved/opened/attempted: {lock_path}") from exc

    output_dir.mkdir(parents=True, exist_ok=False)
    spec_path = output_dir / "F14_V2_FROZEN_BEFORE_OOS.json"

    oos_opened = False

        lock.update({"status": status, "phase": phase_name, "protected_2026_opened": False, **extra})
        write_json(lock_path, lock)
