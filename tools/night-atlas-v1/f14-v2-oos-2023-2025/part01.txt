def sha256_file(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()

def admitted_rows(manifest_path: Path):
    if not manifest_path.is_file():
        raise AdmissionError("manifest absent")
    if "production" in {p.lower() for p in manifest_path.resolve().parts}:
        raise AdmissionError("production input path forbidden")
    if canonical_manifest_sha256(manifest_path) != PINNED_MANIFEST_SHA256:
        raise AdmissionError("manifest hash mismatch")
    manifest = json.loads(manifest_path.read_text(encoding="utf-8"))
    datasets = validate_manifest(manifest)
    duka = datasets["DUKASCOPY_XAUUSD_BID_M1_BI5_2004_2025"]
    if duka["asset"] != "XAUUSD" or duka["timezone"] != "UTC" or duka["granularity"] != "M1":
        raise AdmissionError("F14 V2 requires XAUUSD UTC M1")
    rows = load_dukascopy_index(Path(duka["payload_index"]), duka["payload_index_sha256"])
    for row in rows:
        day = datetime.fromisoformat(row["date"]).replace(tzinfo=UTC)
        if WARMUP_START <= day < END_EXCLUSIVE:
            yield row

class WilderATR:
    def __init__(self, period: int = 14):
        self.period = period
        self.previous_close = None
        self.initial = []
        self.value = None
        self.available_at = None

    def update(self, bar: Bar):
        if self.previous_close is None:
            self.previous_close = bar.close
            self.available_at = bar.available_at
            return
        tr = max(
            bar.high - bar.low,
            abs(bar.high - self.previous_close),
            abs(bar.low - self.previous_close),
        )
        if self.value is None:
            self.initial.append(tr)
            if len(self.initial) == self.period:
                self.value = sum(self.initial) / self.period
        else:
            self.value = ((self.value * (self.period - 1)) + tr) / self.period
        self.previous_close = bar.close
        self.available_at = bar.available_at

class Aggregator:
    def __init__(self, minutes: int):
        self.minutes = minutes
        self.bucket = []

    def push(self, bar: Bar):
        if not self.bucket:
            if bar.timestamp.minute % self.minutes:
                raise AdmissionError(f"M{self.minutes} bucket misaligned")
        elif bar.timestamp - self.bucket[-1].timestamp != timedelta(minutes=1):
            raise AdmissionError(f"M1 gap inside M{self.minutes}")
        self.bucket.append(bar)
        if len(self.bucket) < self.minutes:
            return None
        if len(self.bucket) > self.minutes:
            raise AdmissionError("aggregator overflow")
        out = Bar(
            self.bucket[0].timestamp,
            self.bucket[-1].available_at,
            self.bucket[0].open,
            max(x.high for x in self.bucket),
            min(x.low for x in self.bucket),
            self.bucket[-1].close,
            sum((x.volume or 0.0) for x in self.bucket),
        )
        self.bucket = []
        return out

class Engine:
    def __init__(self):
        self.agg5 = Aggregator(5)
        self.atr5 = WilderATR(14)
        self.m5_hist = []
        self.current_day = None
        self.day_open = None
        self.day_high = None
        self.day_low = None
        self.day_close = None
        self.prev_day = None
        self.prev_m1_timestamp = None
        self.pending = []
        self.completed = []
        self.raw_signals = 0

    def roll_day(self, bar: Bar):
        d = bar.timestamp.date()
        if self.current_day == d:
            return
        if self.current_day is not None and None not in (self.day_high, self.day_low, self.day_close):
            self.prev_day = {
                "high": float(self.day_high),
                "low": float(self.day_low),
                "close": float(self.day_close),
            }
        self.current_day = d
        self.day_open = bar.open
        self.day_high = None
        self.day_low = None
        self.day_close = None

    def update_day(self, bar: Bar):
        self.day_high = bar.high if self.day_high is None else max(self.day_high, bar.high)
        self.day_low = bar.low if self.day_low is None else min(self.day_low, bar.low)
        self.day_close = bar.close

    def after_m5(self, bar: Bar):
        self.m5_hist.append(bar)
        if len(self.m5_hist) > 10000:
            del self.m5_hist[: len(self.m5_hist) - 10000]
        self.atr5.update(bar)

    def detect(self, t: datetime, entry: float):
        if t < OOS_START or t >= END_EXCLUSIVE - timedelta(minutes=HORIZON):
            return
        if not self.prev_day or not self.m5_hist or self.atr5.value is None or self.atr5.value <= 0:
            return
        sb = self.m5_hist[-1]
        if sb.timestamp.date() != t.date():
            return
        if sb.available_at > t or (self.atr5.available_at is not None && self.atr5.available_at > t):
            raise AdmissionError("lookahead")
        pdl = self.prev_day["low"]
        if not (sb.low < pdl && sb.close >= pdl):
            return
        atr = float(self.atr5.value)
        sweep = (pdl - sb.low) / atr
