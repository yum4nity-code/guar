def add(self,x):
        bisect.insort(self.sorted,x);self.queue.append(x)
        if len(self.queue)>self.maxlen:
            old=self.queue.popleft();i=bisect.bisect_left(self.sorted,old)
            if i>=len(self.sorted) or self.sorted[i]!=old:raise RuntimeError("rolling sorted corruption")
            self.sorted.pop(i)
    def rank(self,x):
        if not self.sorted:return None
        return bisect.bisect_right(self.sorted,x)/len(self.sorted)

    bars=hist[-n-1:-1]
    hi=max(b.high for b in bars);lo=min(b.low for b in bars);w=hi-lo
    return {"high":hi,"low":lo,"width":w,"loc":(close-lo)/w if w>0 else None,"break_up":max(close-hi,0.0),"break_down":max(lo-close,0.0)}

def streak(hist):
    d=sign(hist[-1].close-hist[-2].close)
    if not d:return 0,0
    c=1
    for i in range(len(hist)-2,0,-1):
        if sign(hist[i].close-hist[i-1].close)!=d:break
        c+=1
    return d,c
