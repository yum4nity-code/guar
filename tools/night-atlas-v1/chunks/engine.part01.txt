def file_sha256(path:Path)->str:
    h=hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda:f.read(1024*1024),b""): h.update(chunk)
    return h.hexdigest()

def admitted_rows(manifest_path:Path):
    if not manifest_path.is_file(): raise AdmissionError("manifest absent")
    if "production" in {x.lower() for x in manifest_path.resolve().parts}: raise AdmissionError("production input path forbidden")
    if canonical_manifest_sha256(manifest_path)!=PINNED_MANIFEST_SHA256: raise AdmissionError("manifest hash mismatch")
    manifest=json.loads(manifest_path.read_text(encoding="utf-8"))
    datasets=validate_manifest(manifest)
    duka=datasets["DUKASCOPY_XAUUSD_BID_M1_BI5_2004_2025"]
    if duka["asset"]!="XAUUSD" or duka["timezone"]!="UTC" or duka["granularity"]!="M1": raise AdmissionError("Night Atlas requires XAUUSD UTC M1")
    rows=load_dukascopy_index(Path(duka["payload_index"]),duka["payload_index_sha256"])
    for row in rows:
        day=datetime.fromisoformat(row["date"]).replace(tzinfo=UTC)
        if WARMUP_START<=day<END_EXCLUSIVE: yield row

class WilderRSI:
    def __init__(self,period=14):
        self.period=period;self.previous=None;self.gains=deque(maxlen=period);self.losses=deque(maxlen=period)
        self.avg_gain=None;self.avg_loss=None;self.value=None;self.available_at=None
    def update(self,close,available_at):
        if self.previous is None:
            self.previous=close;self.available_at=available_at;return
        d=close-self.previous;g=max(d,0.0);l=max(-d,0.0)
        if self.avg_gain is None:
            self.gains.append(g);self.losses.append(l)
            if len(self.gains)==self.period:
                self.avg_gain=sum(self.gains)/self.period;self.avg_loss=sum(self.losses)/self.period
        else:
            self.avg_gain=((self.avg_gain*(self.period-1))+g)/self.period
            self.avg_loss=((self.avg_loss*(self.period-1))+l)/self.period
        if self.avg_gain is not None and self.avg_loss is not None:
            if self.avg_loss==0:self.value=100.0 if self.avg_gain>0 else 50.0
            else:
                rs=self.avg_gain/self.avg_loss;self.value=100.0-100.0/(1.0+rs)
        self.previous=close;self.available_at=available_at

class WilderATR:
    def __init__(self,period=14):
        self.period=period;self.previous_close=None;self.initial=deque(maxlen=period);self.value=None;self.available_at=None
    def update(self,bar):
        if self.previous_close is None:
            self.previous_close=bar.close;self.available_at=bar.available_at;return
        tr=max(bar.high-bar.low,abs(bar.high-self.previous_close),abs(bar.low-self.previous_close))
        if self.value is None:
            self.initial.append(tr)
            if len(self.initial)==self.period:self.value=sum(self.initial)/self.period
        else:self.value=((self.value*(self.period-1))+tr)/self.period
        self.previous_close=bar.close;self.available_at=bar.available_at

class Aggregator:
    def __init__(self,minutes):self.minutes=minutes;self.bucket=[]
    def push(self,bar):
        if not self.bucket:
            if bar.timestamp.minute%self.minutes: raise AdmissionError(f"M{self.minutes} bucket misaligned")
        elif bar.timestamp-self.bucket[-1].timestamp!=timedelta(minutes=1): raise AdmissionError(f"M1 gap inside M{self.minutes}")
        self.bucket.append(bar)
        if len(self.bucket)<self.minutes:return None
        if len(self.bucket)>self.minutes:raise AdmissionError("aggregator overflow")
        b=Bar(self.bucket[0].timestamp,self.bucket[-1].available_at,self.bucket[0].open,max(x.high for x in self.bucket),min(x.low for x in self.bucket),self.bucket[-1].close,sum((x.volume or 0.0) for x in self.bucket))
        self.bucket=[];return b
