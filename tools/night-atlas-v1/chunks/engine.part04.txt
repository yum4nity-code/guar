,entry,-d,"F15_WEEKEND_GAP","REV",abs(gap)/atr,gap/atr,cr,cp);self.add(t,entry,d,"F15_WEEKEND_GAP","CONT",abs(gap)/atr,gap/atr,cr,cp)
        if self.day_open is not None and sb.timestamp.date()==t.date():
            dev=(sb.close-self.day_open)/atr;d=sign(dev)
            if d and abs(dev)>=DAYOPEN_EXTREME_ATR and d not in self.day_extreme_triggered:self.day_extreme_triggered.add(d);self.add(t,entry,-d,"F16_DAYOPEN_EXTREME","REV",abs(dev),dev,cr,cp)
        if cp is not None and len(self.comp.sorted)>=COMPRESSION_MIN_HISTORY and cp<=COMPRESSION_PERCENTILE:
            if sb.close>chi:self.add(t,entry,1,"F17_COMPRESSION12","CONT",(sb.close-chi)/atr,cp,cr,cp)
            elif sb.close<clo:self.add(t,entry,-1,"F17_COMPRESSION12","CONT",(clo-sb.close)/atr,cp,cr,cp)
    def process_pending(self,bar):
        keep=[]
        for p in self.pending:
            p.age+=1;fav=max((bar.high-p.entry) if p.side>0 else (p.entry-bar.low),0.0);adv=max((p.entry-bar.low) if p.side>0 else (bar.high-p.entry),0.0)
            if fav>p.mfe:p.mfe=fav;p.mfe_minute=p.age
            if adv>p.mae:p.mae=adv;p.mae_minute=p.age
            for level in THRESHOLDS_R:
                if p.threshold_done[level]:continue
                plus=fav>=level*p.risk;minus=adv>=level*p.risk
                if plus or minus:
                    k=str(level).replace(".","p");p.row[f"hit_{k}_winner"]="AMBIGUOUS" if plus and minus else "PLUS" if plus else "MINUS";p.row[f"hit_{k}_minute"]=p.age;p.threshold_done[level]=True
            if p.age in HORIZONS:
                h=p.age;pnl=p.side*(bar.close-p.entry);p.row[f"h{h}_endpoint_price"]=bar.close;p.row[f"h{h}_endpoint_pnl_price"]=pnl;p.row[f"h{h}_endpoint_r"]=pnl/p.risk;p.row[f"h{h}_mfe_price"]=p.mfe;p.row[f"h{h}_mfe_r"]=p.mfe/p.risk;p.row[f"h{h}_mae_price"]=p.mae;p.row[f"h{h}_mae_r"]=p.mae/p.risk;p.row[f"h{h}_minutes_to_mfe"]=p.mfe_minute;p.row[f"h{h}_minutes_to_mae"]=p.mae_minute;p.row[f"h{h}_mfe_before_mae"]=None if p.mfe_minute is None or p.mae_minute is None or p.mfe_minute==p.mae_minute else p.mfe_minute<p.mae_minute
                for c in COSTS:p.row[f"h{h}_r_cost_{c:.2f}"]=(pnl-c)/p.risk
            if p.age>=MAX_HORIZON_MIN:self.writer.writerow(p.row);self.written+=1
            else:keep.append(p)
        self.pending=keep
    def after_m5(self,b):
        self.m5_hist.append(b);self.rsi5.update(b.close,b.available_at);self.atr5.update(b)
        if len(self.m5_hist)>=12 and self.atr5.value and self.atr5.value>0:
            q=list(self.m5_hist)[-12:];self.comp.add((max(x.high for x in q)-min(x.low for x in q))/self.atr5.value)
    def on_m1(self,bar):
        self.roll_day(bar)
        if self.prev_m1_timestamp is not None:
            gap_minutes=int((bar.timestamp-self.prev_m1_timestamp).total_seconds()//60)
            if gap_minutes>1:
                # Mark only positions that existed before this new bar. New Monday-open
                # signals are created after this block and therefore do not inherit the gap.
                for p in self.pending:
                    p.row["outcome_crossed_calendar_gap"]=True
                    p.row["outcome_max_gap_minutes"]=max(int(p.row.get("outcome_max_gap_minutes") or 0),gap_minutes-1)
        if bar.timestamp.minute%5==0:self.detect(bar.timestamp,bar.open)
        self.process_pending(bar);self.rsi1.update(bar.close,bar.available_at);self.atr1.update(bar)
        b5=self.agg5.push(bar)
        if b5:self.after_m5(b5)
        b15=self.agg15.push(bar)
        if b15:self.rsi15.update(b15.close,b15.available_at);self.atr15.update(b15)
        self.update_day(bar);self.prev_m1_timestamp=bar.timestamp

def preflight(manifest,standards):
    rows=list(admitted_rows(manifest))
    for p in (standards/"GUARDIAN_RESEARCH_PROTOCOL_V1.md",standards/"TRADE_OBSERVATION_SCHEMA_V1.json",standards/"guardian_observation_v1.py"):
        if not p.is_file():raise AdmissionError(f"missing research standard: {p}")
    return {"status":"PREFLIGHT_ONLY","id":ID,"days":len(rows),"discovery_start":DISCOVERY_START.isoformat(),"discovery_end_exclusive":END_EXCLUSIVE.isoformat(),"protected_2023_plus_opened":False,"protected_2026_opened":False}

def run(manifest,standards,out):
    if out.exists():raise AdmissionError(f"refusing overwrite: {out}")
    out.mkdir(parents=True);csvp=out/"signals.csv";code_hash=file_sha256(Path(__file__));mh=canonical_manifest_sha256(manifest)
    (out/"family_specs.json").write_text(json.dumps({"schema":1,"id":ID,"families":FAMILY_SPECS,"code_sha256":code_hash,"manifest_sha256":mh,"protected_2023_plus_opened":False,"protected_2026_opened":False},indent=2)+"\n",encoding="utf-8")
    rows=list(admitted_rows(manifest))
    with csvp.open("w",newline="",encoding="utf-8") as f:
        w=csv.DictWriter(f,fieldnames=FIELDNAMES);w.writeheader();atlas=Atlas(w)
        for i,row in enumerate(rows,1):
            for bar in decode_dukascopy_m1_day(row):
                if not WARMUP_START<=bar.timestamp<END_EXCLUSIVE:raise AdmissionError("bar outside hard wall")
                atlas.on_m1(bar)
            if i%100==0 or i==len(rows):print(json.dumps({"status":"PROGRESS","days_done":i,"days_total":len(rows),"signals":atlas.signals,"pending":len(atlas.pending),"date":row["date"]}),flush=True)
        discarded_end_wall=len(atlas.pending);atlas.pending=[]
    summary={"schema":1,"status":"COMPLETE","id":ID,"signals_created":atlas.signals,"signals_written":atlas.written,"discarded_unfinished_at_end":discarded_end_wall,"family_counts":atlas.family_counts,"discovery_start":DISCOVERY_START.isoformat(),"discovery_end_exclusive":END_EXCLUSIVE.isoformat(),"protected_2023_plus_opened":False,"protected_2026_opened":False,"code_sha256":code_hash,"manifest_sha256":mh,"signals_csv_sha256":file_sha256(csvp)}
    (out/"summary.json").write_text(json.dumps(summary,indent=2,sort_keys=True)+"\n",encoding="utf-8");return summary

def main():
    a=argparse.ArgumentParser();a.add_argument("--manifest",required=True,type=Path);a.add_argument("--standards",required=True,type=Path);a.add_argument("--output-dir",required=True,type=Path);a.add_argument("--execute",action="store_true");x=a.parse_args()
